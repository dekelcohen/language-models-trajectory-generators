# Utility: Generate a straight-line trajectory in task space between two 4D poses
def generate_linear_trajectory(start_pose, end_pose, num_points=7):
    """
    Generate a straight-line trajectory in Cartesian space for the end-effector.
    Params:
      - start_pose: [x, y, z, theta]
      - end_pose:   [x, y, z, theta]
      - num_points: number of interpolation points (>=5 recommended)
    Returns:
      - list of [x, y, z, theta] poses
    """
    sx, sy, sz, s_theta = start_pose
    ex, ey, ez, e_theta = end_pose
    traj = []
    for i in range(num_points):
        t = i / (num_points - 1)
        traj.append([
            sx + (ex - sx) * t,
            sy + (ey - sy) * t,
            sz + (ez - sz) * t,
            s_theta + (e_theta - s_theta) * t
        ])
    return traj
